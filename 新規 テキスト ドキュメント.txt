# PDH response
@jax.jit
def pdh_com_stable(f, a, dw, wb, wc, ri, re, ti, Tc):
    w = 2 * jnp.pi * f

    exp_dw = jnp.exp(-2j * dw * Tc)
    exp_w  = jnp.exp(-2j * w * Tc)

    cos_term = jnp.cos(2 * wb * Tc)

    numg = (cos_term - ri * re * exp_dw) * exp_dw

    e1 = jnp.exp(-2j * (dw + wb) * Tc)
    e2 = jnp.exp(-2j * (dw - wb) * Tc)

    deng = (1 - ri * re * e1) * (1 - ri * re * e2)

    gamma = numg / deng

    # ===== A =====
    tmp = (re / ri) * ti**2 * gamma

    A = jnp.real(1 - tmp)

    term1 = -4 * ri**2 * A

    # ===== term2 =====
    exp_wc = jnp.exp(2j * (dw + wc) * Tc)

    B = 1 - tmp

    num = jnp.real(exp_wc * B) - ri * re * A * exp_w

    e3 = jnp.exp(2j * (dw + wc - w) * Tc)
    e4 = jnp.exp(-2j * (dw + wc + w) * Tc)

    den = (1 - ri * re * e3) * (1 - ri * re * e4)

    term2 = 4 * ri * re * ti**2 * exp_w * num / den

    out = a * (term1 + term2) / (1j * f)

    return out

    # One direction
@jax.jit
def pdh_fun_jax1(f, params):
    a00, a01, a002, a012, dw1, dw2, ri, re, ti, te, Tc = params

    return (
        pdh_com_stable(f, -a00, 0.0, wm, 0.0, ri, re, ti, Tc) + pdh_com_stable(f,  a00, 0.0, 0.0, wm, ri, re, ti, Tc) +
        pdh_com_stable(f, -a01, dw1,  wm, 0.0, ri, re, ti, Tc) + 0.5 * pdh_com_stable(f, a01, dw1, 0.0,  wm, ri, re, ti, Tc) + 0.5 * pdh_com_stable(f, a01, dw1, 0.0, -wm, ri, re, ti, Tc))

# Another direction
@jax.jit
def pdh_fun_jax2(f, params):
    a00, a01, a002, a012, dw1, dw2, re, ri, te, ti, Tc = params

    return (
        pdh_com_stable(f, -a002, 0.0, wm, 0.0, ri, re, ti, Tc) + pdh_com_stable(f,  a002, 0.0, 0.0, wm, ri, re, ti, Tc) +
        pdh_com_stable(f, -a012, dw2,  wm, 0.0, ri, re, ti, Tc) + 0.5 * pdh_com_stable(f, a012, dw2, 0.0,  wm, ri, re, ti, Tc) + 0.5 * pdh_com_stable(f, a012, dw2, 0.0, -wm, ri, re, ti, Tc))

# Noise of BLFPI
def BLFPI(f):
    return jnp.sqrt(7e-23**2+1.2e-46*f**(-4)+2.25e-46*f**2)

# Error noise
def Ne1(f, params):
    f = jnp.atleast_1d(f)
    return jnp.ones_like(f) * (7e-23 / jnp.sqrt(2)) * jnp.abs(pdh_fun_jax1(f, params))[0]

def Ne2(f, params):
    f = jnp.atleast_1d(f)
    return jnp.ones_like(f) * (7e-23 / jnp.sqrt(2)) * jnp.abs(pdh_fun_jax2(f, params))[0]

# Suppressed Laser frequency noise
def fnoise(f):
    return 0.1*f**(-1)*1e-4/6e14

def nullbFSR1(params):

  pdh_values = pdh_fun_jax1(freqbfsr, params)

  # Reverting to the more comprehensive null function that incorporates noise
  # This form aims to minimize the combined effect of signal error and noise.
  term_numerator_signal = jnp.abs(1 - Sbfsr / pdh_values)**2*fnoise(freqbfsr)**2
  term_noise_contribution =Ne1(freqbfsr, params)**2/jnp.abs(pdh_values)**2

  NULL_array = (term_numerator_signal + term_noise_contribution) *dfreqbfsr
  return (jnp.sum(NULL_array))

  return (jnp.sum(NULL_array))

@jax.jit
def to_physical(x):

    x = x.reshape(Nset, Nparam)

    return jnp.where(
        log_mask[None, :],
        jnp.exp(x),
        x

@jax.jit
def to_raw(p):

    return jnp.where(
        log_mask[None, :],
        jnp.log(p),
        p
    ).reshape(-1)

@jax.jit
def sample_rayleigh_from_asd(asd, key):

    sigma = asd / jnp.sqrt(jnp.pi / 2)

    key, subkey = jax.random.split(key)

    samples = jax.random.rayleigh(
        subkey,
        scale=sigma,
        shape=asd.shape
    )

    return samples, key

@jax.jit
def residuals_jax(
    x,

    fnoise_sample1,
    fnoise_sample2,
    fnoise_sample3,

    fnoise_sample4,
    fnoise_sample5,
    fnoise_sample6,

    Ne_sample1,
    Ne_sample2,
    Ne_sample3,

    Ne_sample4,
    Ne_sample5,
    Ne_sample6,

    beat_sample1,
    beat_sample2,
    beat_sample3
):

    p = to_physical(x)

    x1 = p[0]
    x2 = p[1]
    x3 = p[2]

    # =========================
    # model PDH
    # =========================
    p11 = pdh_fun_jax1(
        freqbfsr,
        x1
    )

    p12 = pdh_fun_jax2(
        freqbfsr,
        x1
    )

    p21 = pdh_fun_jax1(
        freqbfsr,
        x2
    )

    p22 = pdh_fun_jax2(
        freqbfsr,
        x2
    )

    p31 = pdh_fun_jax1(
        freqbfsr,
        x3
    )

    p32 = pdh_fun_jax2(
        freqbfsr,
        x3
    )

    # =========================
    # response noise
    # =========================
    rn1 = (
        1 - P11_TRUE / p11
    ) * fnoise_sample1

    rn2 = (
        1 - P12_TRUE / p12
    ) * fnoise_sample2

    rn3 = (
        1 - P21_TRUE / p21
    ) * fnoise_sample3

    rn4 = (
        1 - P22_TRUE / p22
    ) * fnoise_sample4

    rn5 = (
        1 - P31_TRUE / p31
    ) * fnoise_sample5

    rn6 = (
        1 - P32_TRUE / p32
    ) * fnoise_sample6

    # =========================
    # early real/imag split
    # =========================
    rn1_real = jnp.real(rn1)
    rn1_imag = jnp.imag(rn1)

    rn2_real = jnp.real(rn2)
    rn2_imag = jnp.imag(rn2)

    rn3_real = jnp.real(rn3)
    rn3_imag = jnp.imag(rn3)

    rn4_real = jnp.real(rn4)
    rn4_imag = jnp.imag(rn4)

    rn5_real = jnp.real(rn5)
    rn5_imag = jnp.imag(rn5)

    rn6_real = jnp.real(rn6)
    rn6_imag = jnp.imag(rn6)

    # =========================
    # manual abs
    # =========================
    abs_p11 = jnp.sqrt(
        jnp.real(p11)**2
        + jnp.imag(p11)**2
    )

    abs_p12 = jnp.sqrt(
        jnp.real(p12)**2
        + jnp.imag(p12)**2
    )

    abs_p21 = jnp.sqrt(
        jnp.real(p21)**2
        + jnp.imag(p21)**2
    )

    abs_p22 = jnp.sqrt(
        jnp.real(p22)**2
        + jnp.imag(p22)**2
    )

    abs_p31 = jnp.sqrt(
        jnp.real(p31)**2
        + jnp.imag(p31)**2
    )

    abs_p32 = jnp.sqrt(
        jnp.real(p32)**2
        + jnp.imag(p32)**2
    )

    # =========================
    # electronics noise
    # =========================
    re1 = Ne_sample1 / abs_p11

    re2 = Ne_sample2 / abs_p12

    re3 = Ne_sample3 / abs_p21

    re4 = Ne_sample4 / abs_p22

    re5 = Ne_sample5 / abs_p31

    re6 = Ne_sample6 / abs_p32

    # =========================
    # concatenate
    # =========================
    res = jnp.concatenate([

        rn1_real,
        rn1_imag,

        rn2_real,
        rn2_imag,

        rn3_real,
        rn3_imag,

        rn4_real,
        rn4_imag,

        rn5_real,
        rn5_imag,

        rn6_real,
        rn6_imag,

        re1,
        re2,
        re3,
        re4,
        re5,
        re6,

        beat_sample1,
        beat_sample2,
        beat_sample3
    ])

    return (
        res
        * jnp.sqrt(dfreqbfsr)
        / init_res
    )

def make_wrappers(

    fnoise_sample1,
    fnoise_sample2,
    fnoise_sample3,

    fnoise_sample4,
    fnoise_sample5,
    fnoise_sample6,

    Ne_sample1,

# =========================
# block wrapper
# =========================
def make_block_wrapper(
    residuals_np_full,
    jacobian_np_full,
    x_fixed,
    block_id,
    work_array
):

    start = block_id * Nparam
    end = start + Nparam

    def residuals_block(x_block):

        work_array[:] = x_fixed
        work_array[start:end] = x_block

        return residuals_np_full(work_array)

    def jacobian_block(x_block):

        work_array[:] = x_fixed
        work_array[start:end] = x_block

        J = jacobian_np_full(work_array)

        return J[:, start:end]

    return residuals_block, jacobian_block

# =========================
# single optimization
# =========================
def run_single_opt(
    x0,
    residuals_np,
    jacobian_np,
    bounds
):

    return least_squares(
        fun=residuals_np,
        x0=x0,
        jac=jacobian_np,
        bounds=bounds,
        method="trf",
        x_scale='jac',
        ftol=1e-15,
        xtol=1e-15,
        gtol=1e-15,
        max_nfev=10,
        verbose=0
    )

# =========================
# one chain
# =========================
def run_chain(
    chain_id,
    true_params,
    bounds,
    bounds_blocks,
    residuals_np_full,
    jacobian_np_full
):

    print(f"chain {chain_id} start")

    try:

        init_noise = 0.01 * np.random.randn(
            Nset,
            Nparam
        )

        init_params = (
            np.asarray(true_params)
            * np.exp(init_noise)
        )

        x_init = np.array(
            np.asarray(
                to_raw(
                    jnp.array(init_params)
                )
            ),
            dtype=np.float64
        )

        lower_bounds = bounds[0]
        upper_bounds = bounds[1]

        eps = 1e-12

        x_init = np.clip(
            x_init,
            lower_bounds + eps,
            upper_bounds - eps
        )

        x_current = np.array(
            x_init,
            dtype=np.float64,
            copy=True
        )

        work_array = np.empty_like(
            x_current
        )

        block_wrappers = []

        for block_id in range(Nset):

            block_wrappers.append(

                make_block_wrapper(
                    residuals_np_full,
                    jacobian_np_full,
                    x_current,
                    block_id,
                    work_array
                )
            )

        # =========================
        # coarse optimization
        # =========================
        for coord_iter in range(
            Ncoord_coarse
        ):

            for block_id in range(Nset):

                start = (
                    block_id
                    * Nparam
                )

                end = start + Nparam

                residuals_block, jacobian_block = (
                    block_wrappers[block_id]
                )

                x0_block = x_current[
                    start:end
                ]

                result = run_single_opt(
                    x0_block,
                    residuals_block,
                    jacobian_block,
                    bounds_blocks[block_id]
                )

                x_current[start:end] = (
                    result.x
                )

        r = residuals_np_full(
            x_current
        )

        coarse_cost = (
            0.5 * np.dot(r, r)
        )

        print(
            f"chain {chain_id} coarse cost = "
            f"{coarse_cost}"
        )

        if coarse_cost > COARSE_COST_THRESHOLD:

            print(
                f"chain {chain_id} rejected"
            )

            return None

        # =========================
        # fine optimization
        # =========================
        zero_reduction_count = 0

        cost_tol = 1e-30

        prev_total_cost = (
            coarse_cost
        )

        for coord_iter in range(
            Ncoord_coarse,
            Ncoord
        ):

            for block_id in range(Nset):

                start = (
                    block_id
                    * Nparam
                )

                end = start + Nparam

                residuals_block, jacobian_block = (
                    block_wrappers[block_id]
                )

                x0_block = x_current[
                    start:end
                ]

                result = run_single_opt(
                    x0_block,
                    residuals_block,
                    jacobian_block,
                    bounds_blocks[block_id]
                )

                x_current[start:end] = (
                    result.x
                )

            r = residuals_np_full(
                x_current
            )

            current_total_cost = (
                0.5 * np.dot(r, r)
            )

            cost_reduction = (
                prev_total_cost
                - current_total_cost
            )

            if np.abs(cost_reduction) < cost_tol:

                zero_reduction_count += 1

            else:

                zero_reduction_count = 0

            if zero_reduction_count >= Nset:

                break

            prev_total_cost = (
                current_total_cost
            )

        final_cost = (
            current_total_cost
        )

        params_opt = np.asarray(
            to_physical(x_current)
        )

        return {

            "params_opt": params_opt,
            "cost": final_cost
        }

    except Exception as e:

        print(e)

        return None

        end = start + Nparam

        bounds_blocks.append(
            (
                bounds[0][start:end],
                bounds[1][start:end]
            )
        )

    #key = jax.random.PRNGKey(0)
    seed = int(time.time_ns())
    key = jax.random.PRNGKey(seed)

    results_log = []

    best_cost_global = np.inf
    best_result_global = None

    successful_trial = 0

    # =========================
    # successful trial loop
    # =========================
    while successful_trial < Ntrial:

        print("\n====================")
        print(f"TRIAL {successful_trial}")
        print("====================")

        (
            fnoise_full1,
            fnoise_full2,
            fnoise_full3,

            fnoise_full4,
            fnoise_full5,
            fnoise_full6,

            Ne_full1,
            Ne_full2,
            Ne_full3,

            Ne_full4,
            Ne_full5,
            Ne_full6,

            beat_full1,
            beat_full2,
            beat_full3,

            key

        ) = generate_noise(key)

        # =========================
        # mask
        # =========================
        fnoise_sample1 = (
            fnoise_full1[mask]
        )

        fnoise_sample2 = (
            fnoise_full2[mask]
        )

        fnoise_sample3 = (
            fnoise_full3[mask]
        )

        fnoise_sample4 = (
            fnoise_full4[mask]
        )

        fnoise_sample5 = (

        )

        fnoise_sample6 = (
            fnoise_full6[mask]
        )

        Ne_sample1 = (
            Ne_full1[mask]
        )

        Ne_sample2 = (
            Ne_full2[mask]
        )

        Ne_sample3 = (
            Ne_full3[mask]
        )

        Ne_sample4 = (
            Ne_full4[mask]
        )

        Ne_sample5 = (
            Ne_full5[mask]
        )

        Ne_sample6 = (
            Ne_full6[mask]
        )

        beat_sample1 = (
            beat_full1[mask]
        )

        beat_sample2 = (
            beat_full2[mask]
        )

        beat_sample3 = (
            beat_full3[mask]
        )

        residuals_np_full, jacobian_np_full = (
            make_wrappers(

                fnoise_sample1,
                fnoise_sample2,
                fnoise_sample3,

                fnoise_sample4,
                fnoise_sample5,
                fnoise_sample6,

                Ne_sample1,
                Ne_sample2,
                Ne_sample3,

                Ne_sample4,
                Ne_sample5,
                Ne_sample6,

                beat_sample1,
                beat_sample2,
                beat_sample3
            )
        )

        # =========================
        # warmup compile
        # =========================
        dummy_x = np.asarray(
            to_raw(true_params),
            dtype=np.float64
        )

        residuals_np_full(dummy_x)
        jacobian_np_full(dummy_x)

        # =========================
        # retry until accepted
        # =========================
        accepted = False

        while not accepted:

            results_parallel = Parallel(
                n_jobs=Nparallel,
                backend="threading"
            )(
                delayed(run_chain)(
                    chain_id,
                    true_params,
                    bounds,
                    bounds_blocks,
                    residuals_np_full,
                    jacobian_np_full
                )
                for chain_id in range(
                    Nparallel
                )
            )

            results_parallel = [
                r for r in results_parallel
                if r is not None
            ]

            if len(results_parallel) == 0:

                print(
                    "All chains rejected. "
                    "Restart same trial."
                )

                continue

            accepted = True

        # =========================
        # best chain
        # =========================
        best_trial = min(
            results_parallel,
            key=lambda r: r["cost"]
        )

        print("\nBEST IN THIS TRIAL")

        print(
            "cost =",
            best_trial["cost"]
        )

        # =========================
        # save
        # =========================
        results_log.append({

            "params_opt":
                best_trial["params_opt"],

            "cost":
                best_trial["cost"],

            "fnoise_sample1":
                np.asarray(fnoise_full1),

            "fnoise_sample2":
                np.asarray(fnoise_full2),

            "fnoise_sample3":
                np.asarray(fnoise_full3),

            "fnoise_sample4":
                np.asarray(fnoise_full4),

            "fnoise_sample5":
                np.asarray(fnoise_full5),

            "fnoise_sample6":
                np.asarray(fnoise_full6),

            "Ne_sample1":
                np.asarray(Ne_full1),

            "Ne_sample2":

            best_result_global = (

            )

        successful_trial += 1

    # save arrays

    params_all = np.array([

    np.savez_compressed(
